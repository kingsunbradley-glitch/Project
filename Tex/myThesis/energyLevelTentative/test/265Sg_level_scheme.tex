% ============================================================

% Decay direction is from right to left.
\def\xAxis{0.00}

    (\xAxis,{\yTwoHundred+0.15})

\node[font=\large,rotate=90]
    at (\xAxis-0.68,{(\yZero+\yTwoHundred)/2})
