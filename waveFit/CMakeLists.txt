cmake_minimum_required(VERSION 3.16)
project(wave_reader LANGUAGES CXX)
find_package(ROOT REQUIRED COMPONENTS Core RIO Hist Graf Gpad Tree Matrix MathCore)
add_executable(wave_reader src/main.cpp src/RootCanvasReader.cpp src/Analysis.cpp src/Output.cpp)
target_include_directories(wave_reader PRIVATE include)
target_compile_features(wave_reader PRIVATE cxx_std_17)
if(ROOT_CXX_STANDARD)
  set_property(TARGET wave_reader PROPERTY CXX_STANDARD ${ROOT_CXX_STANDARD})
endif()
target_compile_options(wave_reader PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(wave_reader PRIVATE ROOT::Core ROOT::RIO ROOT::Hist ROOT::Graf ROOT::Gpad ROOT::Tree ROOT::Matrix ROOT::MathCore)
